  P := nativeProduct>0;

  CF := frontierConstant=(2*pi)^3*P;

  H_P(nu) := {a:CriticalSobolev 1 |
    CriticalSigmaZero a AND P*h(a)<nu}.

  P*h(a)<nu
    <-> p(a)<[sqrt(2)*(2*pi)^3/K2]*nu.

  !!!! FRONTIER_H12_PRODUCT_ADAPTER_CHAIN:
    forall nu>0, forall a:CriticalSobolev 1,
      CriticalSigmaZero a
      -> p(a)<[sqrt(2)*(2*pi)^3/K2]*nu
      -> CriticalDatumAX6 nu a
         AND exists U:CriticalRegularSolution nu a,
           CriticalLerayHopf nu (criticalInitialPhysicalL2 a)
             U.lerayVelocity U.lerayGradient
           AND U.lerayHopfUnique
           AND PositiveSmoothNSE nu U.flow.velocity U.flow.pressure
           AND GlobalProductQuantitativeBounds(nu,a,U).

  CriticalRegularSolution entails:
    U.critical in C([0,infinity);CriticalSobolev 1);
    U.critical 0=a;
    forall t>=0, ||U.critical t||<=||a||;
    positive-time physical/Fourier coefficient agreement.

  U.lerayHopfUnique:
    forall v G, CriticalLerayHopf nu (criticalInitialPhysicalL2 a) v G
    -> forall t>0, v(t)=U.flow.velocity(t) a.e.[volume.restrict unitCube].

  delta := nu-P*||a||>0;
  DP := ||a||^2/(2*delta);
  VP := (nu+criticalTransportConstant*||a||)^2*||a||^2/(2*delta).

  GlobalProductQuantitativeBounds:
    W:=U.flow.threeHalfState in L^2((0,infinity);CriticalSobolev 3),
      integral_0^infinity ||W(t)||^2dt<=DP;

      integral_0^infinity ||Z(t)||^2dt<=VP;

    forall T>0,

        <= [||a||^4-||U.critical T||^4]/(4*delta).
