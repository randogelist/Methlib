  ∀ (ν c : ℝ) (m : Vec) (B : Velocity), MagneticPath B →
    NSEMaxwell.Bridge.CanonicalSmoothness ν c m B

H4 = A_TIER
NSEMaxwell.HodgeProof.h4_electric_hodge :
  ∀ (ν : ℝ) (m : Vec) (B : Field), MagneticData B →
    G ν m B + NSEMaxwell.Classical.gradient (headPotential ν m B)
      = K (spatialCurl (G ν m B))

SCOPE
-----
Data a = ContDiff ℝ ∞ a ∧ SpatialPeriodic a ∧ divergence a = 0
MagneticData B = Data B ∧ spatialMean B = 0
MagneticPath B = SmoothVelocity B ∧ PeriodicVelocity B
  ∧ (∀ t ≥ 0, divergence (B t) = 0 ∧ spatialMean (B t) = 0)

SmoothVelocity B = ContDiffOn ℝ ∞ (fun z => B z.1 z.2)
  {z : ℝ × Vec | 0 ≤ z.1}
SmoothPressure p = ContDiffOn ℝ ∞ (fun z => p z.1 z.2)
  {z : ℝ × Vec | 0 ≤ z.1}

CanonicalSmoothness ν c m B =
  SmoothVelocity (canonicalVelocity m B)
  ∧ SmoothPressure (canonicalPressureTrajectory ν m B)
  ∧ (∀ t ≥ 0, divergence (canonicalVelocity m B t) = 0)
